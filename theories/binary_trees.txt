order v 7 : vertex

let parent : vertex -> vertex
let left : vertex -> vertex
let right : vertex -> vertex

var x, y : vertex

-- Ordering constraints

---- There is an arbitrary total ordering over vertices
not before (x, x)
before (x, x.next)
before (x, y) -> before (x, y.next)

---- No vertex has a directed edge pointing to an edge 'before' it
edge (x, y), before (y, x) -> False

-- Relationship between the edge and parent relations

---- The converse of the edge relation is the parent relation
not root (y), parent (y, x) -> edge (x, y)
edge (x, y), not parent (y, x) -> False

-- The root is unique
root (v[1])
root (x) -> not root (x.next)
not root (x) -> not root (x.next)
not root (x), parent (x, x) -> False

-- The root is its own parent
root (x) -> parent (x, x)

-- A vertex is either a leaf or a node
node (x) & leaf (x)
node (x), leaf (x) -> False

-- A vertex is either a right or left vertex
right (x) & left (x)
right (x), left (x) -> False

-- If a vertex is a left/right vertex, its parent's left/right is that vertex
not root(y), parent (y, x), left (y) -> left (x, y)
not root(y), parent (y, x), right (y) -> right (x, y)

-- 
node(x), left (x, y) -> parent (y, x)
node(x), right (x, y) -> parent (y, x)

-- Right and left vertices are distinct
node (x), right (x, y), left (x, y) -> False

-- Leaves have no out-edges
leaf (x), edge (x, y) -> False

-- Leaves are their own left and right
leaf (x) -> right (x, x)
leaf (x) -> left (x, x)

-- Leaves cannot be parents
parent (y, x), leaf (x) -> False

-- Nodes are never their own left and right, and their left and right must be distinct
not root (x), node (x), right (x, x) -> False
not root (x), node (x), left (x, x) -> False

-- Left and right vertices of x must have x as a parent 
not root(x), node (x), right (x, y), not parent (y, x) -> False
not root(x), node (x), left (x, y), not parent (y, x) -> False

-- The root cannot be its own left and right
root (x), left (x, x) -> False
root (x), right (x, x) -> False

-- The root is always a left node
root (x) -> left (x)
